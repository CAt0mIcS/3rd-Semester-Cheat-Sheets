\subsection{Grundlagen}

\begin{sectionbox}
    \subsubsection{Ergebnis}
    Ausgang, welchen ich messe/vorfinde/beobachte nach Durchführung des Experiments.

    \subsubsection{Ergebnismenge $\Omega$}
    Alle möglichen Ausgänge eines Zufallsexperiments.
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Ereignis $A$}
    Teilmenge von $\Omega$, die eine bestimmte Eigenschaft beschreibt (z.B. Augenzahl gerade)

    \subsubsection{(konsistente) Ereignismenge $\mathbb F$ ($\sigma$-Algebra)}
    Die Menge aller Ereignisse, die im Wahrscheinlichkeitsmodell auftreten können (muss nicht alle möglichen Teilmengen von $\Omega$ enthalten, solange die unteren Bedingungen erfüllt sind)
    \begin{itemize}
        \item $\Omega \in \mathbb F$
        \item $A \in \mathbb F \Rightarrow A^C \in \mathbb F$
        \item $A_i\in\mathbb F \Rightarrow A_1\cup A_2 \cup ... \in \mathbb F$
    \end{itemize}

\begin{sectionbox}
    \subsubsection{Partitionierung von $\Omega$}
    $\Omega$ wird in disjunkte Teilmengen $B_i$ eingeteilt, wobei $\bigcup_{i=1}^n B_i=\Omega$ \\

\begin{sectionbox}
    \subsubsection{Wahrscheinlichkeitsmaß $P$}
    $P$ ist eine Funktion $P : \mathbb F \rightarrow [0,1] \subset \mathbb R$ mit folgenden Eigenschaften
    \begin{itemize}
        \item $P(\emptyset) = 0, P(\Omega) = 1, P(A)\in[0,1]$
        \item $P(A\cup B) = P(A) + P(B) - P(A\cap B)$
    \end{itemize}

    Weitere Eigenschaften
    \begin{itemize}
        \item $P(A\setminus B)=P(A)-P(A\cap B)$
    \end{itemize}

\end{sectionbox}
