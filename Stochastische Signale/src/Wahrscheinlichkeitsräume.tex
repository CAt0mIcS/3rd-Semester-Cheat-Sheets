\begin{sectionbox}
    \subsubsection{De-Morgan}
    $(A^C \cup B^C)^C=A\cap B$
\end{sectionbox}


\begin{sectionbox}
    \subsubsection{Wahrscheinlichkeitsmaß $P$}
    $P$ ist eine Funktion $P : \mathbb F \rightarrow [0,1] \subset \mathbb R$ mit folgenden Eigenschaften
    \begin{itemize}
        \item $P(\emptyset) = 0, P(\Omega) = 1, P(A)\in[0,1]$
        \item $P(A\cup B) = P(A) + P(B) - P(A\cap B)$
    \end{itemize}

    Weitere Eigenschaften
    \begin{itemize}
        \item $P(A\setminus B)=P(A)-P(A\cap B)$
    \end{itemize}

\end{sectionbox}
