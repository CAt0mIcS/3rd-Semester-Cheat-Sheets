\section{Thermische Festkörpereigenschaften}
\begin{emphbox}
    \begin{description}
        \item[Anzahl an Atomen] $N = N_An$
        \item[Gaskonstante] $R$
        \item[Kelvin-Boltzmann-Konstante] $k_B$
        \item[Avogadro-Konstante] $N_A$
        \item[Stoffmenge in mol] $n$
    \end{description}
\end{emphbox}

\begin{sectionbox}
    \subsection{Spezifische Wärme(kapazität)}

    \subsubsection{Wärmekapazität}
    Erforderliche Wärmemenge (innere Energie), um die Temperatur eines Festkörpers um $1 K$ zu erhöhen
    \begin{equation*}
        C=\frac{\Delta U}{\Delta T}
    \end{equation*}

    \subsubsection{spezifische Wärmekapazität}
    \begin{equation*}
        c=\frac{1}{m} C
    \end{equation*}

    \subsubsection{Molwärme - Dulong-Petit-Gesetz}
    \begin{equation*}
        c_m = \frac{1}{n}C=3N_ak_B=3R=24,9\frac{\text{J}}{\text{mol K}}
    \end{equation*}
    Für hohe Temperaturen gilt diese Näherung (dass $c_m=$ const.). Für niedrigere Temperaturen muss man quantenmechanische Phänomene betrachten.

    \subsubsection{Debye-Näherung}
    sagt die Temperaturabhängigkeit der Wärmekapazität sowohl im Niedrig- als auch im Hochtemperaturlimes korrekt voraus.
    \begin{equation*}
        k_B\Theta = \hbar \omega_D
    \end{equation*}
    mit $\Theta$: materialspezifischer Parameter (Debye-Temperatur) und \\
    $\omega_D$: Debyesche Grenzfrequenz $\approx$ Frequenz der optischen Phononen
\end{sectionbox}

\begin{sectionbox}
    \subsection{Wärmeleitfähigkeit}
    Ein Material besitzt genau dann eine hohe Wärmeleitfähigkeit, wenn es eine hohe spezifische Wärme und eine geringe Streuung an Defekten oder anderen Phononen aufweist.
    \begin{equation*}
        \lambda = C\frac{v^2}{3}\tau = \frac{1}{3}Cvl
    \end{equation*}
    mit $\lambda$: Wärmeleitfähigkeit pro Volumen, $l$: mittlere freie Weglänge, $v$: Geschwindigkeit der Teilchen im Festkörper, $\tau$: Streuzeit
\end{sectionbox}


\section{Elektronen im Festkörper}
\subsection{Elektronengas}
\begin{itemize}
    \item Valenzelektronen in Metallen quasi frei beweglich
    \item Bewegung der Elektronen in einem Kasten mit unendlich hohen Potentialwänden (Metalloberfläche ist Begrenzung)
    \item Für großes $L$ liegen die Energieniveaus verschiedener $n$ sehr nahe $\Rightarrow$ Leitungsbänder ähnlicher Energienivenaus, dazwischen verbotene Zonen
    \item \begin{equation*}
              E_{n_x,n_y,n_z}=\frac{\hbar^2\pi^2}{2mL^2}(n_x^2+n_y^2+n_z^2)
          \end{equation*}
\end{itemize}

\subsection{Zustandsdichte}
Beschreibt, wie dicht erlaubte Zustände (Leitungsbänder) aneinander gepackt werden können.
\begin{equation*}
    n = \int_0^\infty\underbrace{\left(\frac{2m}{\hbar^2}\right)^{\frac{3}{2}}\frac{\sqrt{E}}{2\pi^2}}_{\text{energet. Zustandsdichte pro Volumen } D(E)}f(T, E)\diff E
\end{equation*}

\subsection{Fermiverteilung $f(T, E)$}
Besetzungswahrscheinlichkeit für Elektronen im Gleichgewicht. Nähert sich für $T=0$ einer Stufenfunktion an.
\begin{equation*}
    f(E,T)=\frac{1}{e^{\frac{E-E_F}{k_BT}} + 1}
\end{equation*}
mit $E_F$: Fermienergie. Für $T=0K$ gilt:
\begin{equation*}
    n=\int_0^{E_F}\left(\frac{2m}{\hbar^2}\right)^{\frac{3}{2}}\frac{\sqrt{E}}{2\pi^2}\diff E \Longrightarrow E_F=\frac{\hbar^2}{2m}(3\pi^2n)^{\frac32}
\end{equation*}

\subsection{Thermische Anregung}
Im quantenmechanischen werden elektronische Prozesse (z.B. Stromtransport, thermisches Aufheizen) als energetische Anregung von Elektronen aufgefasst.
Hierbei wird das Elektron auf ein höheres Energieniveau gebracht. 
Es können jedoch nur Elektronen nahe $E_F$ exitiert werden. 
Für $E<E_F$ gibt es keine freien Zustände und für $E>E_F$ gibt es keine Elektronen mehr. 
$n_{eff}$ beschreibt die Elektronendichte, die an thermischen Exitierungsprozessen teilnehmen können
\begin{equation*}
    n_{eff} \approx 2D(E_F)k_BT \approx \frac{3n}{2E_F}
\end{equation*}
\begin{equation*}
    C_{el}\approx\frac{6nk_B^2T}{E_F} \quad v_F=\sqrt{\frac{2E_F}{m}}
\end{equation*}
