\begin{sectionbox}
    \subsection{Fourierpolynom}
    \begin{description}
        \item[Periode] betragsmäßig kleinste Zahl, wo sich die Funktion $\forall t$ wiederholt,
              z.B. $f(t)=\cos(k\omega t) + i\sin(k\omega t)$ wiederholt sich für $|k|\omega T_k=2\pi \Rightarrow T_k=\frac{2\pi}{|k|\omega}$
        \item[T-periodisch] Falls $f(t + T)=f(t) \quad \forall t\in\mathbb R$
        \item[Kreisfrequenz] $\omega = \frac{2\pi}{T}$ mit Periode $T$
    \end{description}

    Sei $\omega > 0, c_k \in \mathbb C, -n \le k \le n, n \in \mathbb N$. Dann heißt
    \begin{equation*}
        S_n(t)=\sum_{k=-n}^{n}c_ke^{ik\omega t}, t\in\mathbb R
    \end{equation*}
    Fourierpolynom bzw. trigonometrisches Polynom.

    \subsubsection{Eigenschaften}
    \begin{itemize}
        \item $T_1$ ist die größte auftretende Periode aller Summanden $\Rightarrow S_n$ ist $T_1$-periodisch
        \item $T_k= \frac{T_1}{|k|}$
        \item $\omega$ ist für alle Summanden gleich, $k$ variiert
    \end{itemize}
\end{sectionbox}

\subsection{Fourierreihe}
Jedes Fourierpolynom ist gleichzeitig eine Fourierreihe, wenn man alle nicht benötigten $c_k=0$ setzt.
\begin{emphbox}
    \begin{equation*}
        S_f(t)=\lim_{n\rightarrow\infty}S_n(t)=\sum_{k\in\mathbb Z}c_ke^{ik\omega t}=\sum_{k=-\infty}^{\infty}c_ke^{ik\omega t}
    \end{equation*}
\end{emphbox}

\begin{sectionbox}
    \subsubsection{Bestimmung der Fourierkoeffizienten}
    Sei eine $T$-periodische Funktion $f: \mathbb R \rightarrow \mathbb C$ gegeben und $k\in\mathbb Z$. Dann heißt
    \begin{equation*}
        c_k=\frac{1}{T}\int_a^{a+T} f(t)e^{-ik\omega t}\diff t, \quad a \text{ bel.}
    \end{equation*}
    der $k$-te Fourierkoeffizient von $f$. \\
    \textbf{Fourierspektrum} Familie $\{c_k\}_{k\in\mathbb Z}$ aller Fourierkoeffizienten. Beschreibt Amplituden und Phasenverschiebungen der Teilschwingungen in $f$.

    \subsubsection{Konvergenz der Fourierreihe}
    \begin{sectionbox}
        $f:\mathbb R \rightarrow \mathbb C$ ist stückweise stetig, falls $\forall t \in \mathbb R$ die rechts- und linksseitigen Grenzwerte existieren und $f$ auf jedem beschränkten Interval nur in endlich vielen Punkten unstetig ist (Sprungstellen)
    \end{sectionbox}

    \begin{sectionbox}
        $f$ stückweise stetig ist stückweise stetig differenzierbar, falls $f$ auf jedem beschränkten Intervall nur in endlich vielen Punkten nicht stetig differenzierbar ist und die rechts- und linksseitigen Grenzwerte von $f'$ auch existieren:
        \begin{equation*}
            f'(t^{\pm}) = \lim_{h\rightarrow 0^{\pm}}f'(t\pm h)
        \end{equation*}
    \end{sectionbox}

    Sei $f$ stückweise stetig differenzierbar:
    \begin{description}
        \item[Punktweise Konvergenz] Fourierreihe konvergiert $\forall t$ gegen $\frac{f(t^-) + f(t^+)}{2}$
        \item[Gleichmäßige Konvergenz] Auf jedem abgeschlossenen Intervall ohne Sprungstellen konvergiert die Fourierreihe gegen $f$
    \end{description}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Umrechnung zu sin-/cos-Darstellung}
    \begin{emphbox}
        \begin{equation*}
            S_f(t)=\frac{a_0}{2}+\lim_{n\rightarrow\infty}\sum_{k=1}^{n}(a_k\cos(k\omega t) + b_k\sin(k\omega t))
        \end{equation*}
    \end{emphbox}
    mit $k\in\mathbb N$:
    \begin{equation*}
        a_0=2c_0 \quad a_k=c_k+c_{-k} \quad b_k=i(c_k-c_{-k})
    \end{equation*}
    bzw.
    \begin{align*}
        a_k=\frac{2}{T}\int_0^Tf(t)\cos(k\omega t)\diff t \quad k \in \mathbb N_0 \\
        b_k=\frac{2}{T}\int_0^Tf(t)\sin(k\omega t)\diff t \quad k \in \mathbb N
    \end{align*}

    Falls $f$ reel: $c_{-k} = \overline{c_k}$,
    \begin{equation*}
        a_0=2c_0 \quad a_k=2\Re{c_k} \quad b_k=-2\Im{c_k}
    \end{equation*}
\end{sectionbox}

\begin{sectionbox}
    Bei $T$-periodischem $f$: \\
    $f$ gerade $\Leftrightarrow f(\frac{T}{2}+t) = f(\frac{T}{2}-t)$ \\
    $f$ ungerade $\Leftrightarrow f(\frac{T}{2}+t) = -f(\frac{T}{2}-t)$

    Seien $f,g:\mathbb R \rightarrow\mathbb C \; T$-periodisch. Sei $\{c_k\}_{k\in\mathbb Z}$ das Fourierspektrum von $f$ und $\{d_k\}_{k\in\mathbb Z}$ von $g$
    \subsubsection{Eigenschaften}
    \begin{itemize}
        \item $f$ gerade $\Rightarrow c_k=c_{-k}$ bzw. $b_k=0 \; \forall k \in\mathbb N$. Fourierreihe besteht nur aus $\cos$-Termen
        \item $f$ ungerade $\Rightarrow c_k=-c_{-k}$ bzw. $a_k=0 \; \forall k \in\mathbb N_0$. Fourierreihe besteht nur aus $\sin$-Termen
        \item Das Fourierspektrum von $\alpha f(t) + \beta g(t)$ ist $\{\alpha c_k + \beta d_k\}_{k\in\mathbb Z}$
        \item Das Fourierspektrum von $\overline{f(t)}$ ist $\{\overline{c_{-k}}\}_{k\in\mathbb Z}$
        \item Das Fourierspektrum von $f(-t)$ ist $\{c_{-k}\}_{k\in\mathbb Z}$
        \item Das Fourierspektrum von $f(t + a)$ (Verschiebung im Zeitbereich) ist $\{e^{ik\omega a}c_k\}_{k\in\mathbb Z}$
        \item Das Fourierspektrum von $e^{in\omega t}f(t) \quad n\in\mathbb Z$ (Amplitudenmodulation) ist $\{c_{k-n}\}_{k\in\mathbb Z}$
        \item Das Fourierspektrum von $f(\gamma t)$ (Streckung) ist $\{c_k\}_{k\in\mathbb Z}$
        \item Sei $f$ stetig und $f'$ stückweise stetig: Das Fourierspektrum von $f'(t)$ ist $\{ik\omega c_k\}_{k\in\mathbb Z}$
        \item Sei $f$ stückweise stetig mit $c_0=0$: Das Fourierspektrum von $F(t)=\int_0^t f(s)\diff s$ ist
              \begin{equation*}
                  s_k=\begin{cases}
                      -\frac{1}{T}\int_0^Ttf(t)\diff t & k=0    \\
                      \frac{c_k}{ik\omega}             & k\ne 0
                  \end{cases}
              \end{equation*}
        \item Sei $f$ stückweise stetig und $N < \infty$ Sprungstellen auf $[0,T)$ und dazwischen stetig diffb.. $f'$ außerhalb der Sprungstellen hat das Fourierspektrum \begin{equation*}
                  d_k=ik\omega c_k-\frac{1}{T}\sum_{j=1}^{N}(f(t_j^+) - f(t_j^-))e^{-ik\omega t_j}
              \end{equation*}
    \end{itemize}

    Falls $f:\mathbb R \rightarrow \mathbb R$ gilt
    \begin{itemize}
        \item $c_{-k}=\overline{c_k}$ \; und $\; a_n,b_n\in\mathbb R$
        \item $f$ gerade $\Rightarrow c_k\in\mathbb R$
        \item $f$ ungerade $\Rightarrow c_k\in\mathbb C\setminus\mathbb R$
    \end{itemize}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Eigenschaften des Fourierspektrums}
    \begin{itemize}
        \item $f$ stückweise stetig mit endlich vielen Sprungstellen auf $[0,T] \Rightarrow |c_k| \le C$ (Fourierkoeffizienten beschränkt)
        \item $f$ stückweise stetig mit endlich vielen Sprungstellen auf $[0,T]$ und dazwischen stetig diffb. $\Rightarrow$ Fourierkoeffizienten von $f'$ beschränkt $|d_k| \le M \Rightarrow |c_k| \le \frac{C}{|k|}$
        \item $f$ stetig mit stetigen Ableitungen $f', f'', f^{(m-2)}$, $f^{(m-1)}$ stückweise stetig diffb. $\Rightarrow |c_k| \le \frac{C}{|k|^m}$
    \end{itemize}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Orthogonalreihenrelationen}
    Für $k_1, k_2 \in \mathbb Z$ gilt:
    \begin{equation*}
        \frac{1}{T}\int_0^Te^{ik_1\omega t}e^{-ik_2\omega t}\diff t = \delta_{k_1,k_2}=\begin{cases}
            1, & k_1=k_2    \\
            0, & k_1\ne k_2
        \end{cases}
    \end{equation*}
    $\Rightarrow f_k(t)=e^{ik\omega t}$ hat Fourierspektrum $c_{k'}=\delta_{k,k'} \; \forall k' \in \mathbb Z$
\end{sectionbox}


\subsection{Faltung}
\begin{emphbox}
    Seien $f,g : \mathbb R \rightarrow\mathbb C$ T-periodisch und stückweise stetig
    \begin{equation*}
        (f * g)(t) = \frac{1}{T}\int_0^T f(t-s)g(s)\diff s
    \end{equation*}
\end{emphbox}
Rechenregeln wie Multiplikation

\begin{sectionbox}
    \subsubsection{Eigenschaften}
    \begin{itemize}
        \item $(e^{ik\omega t} * f)(t)=c_ke^{ik\omega t} \Rightarrow k$-te Teilschwingung der Fourierreihe von $f$
        \item $(f * g)(t)$ ist stetig
        \item Das Fourierspektrum von $(f * g)(t)$ ist $\{c_kd_k\}_{k\in\mathbb Z}$
    \end{itemize}
\end{sectionbox}


\subsection{LTI-Systeme}
\begin{sectionbox}
    \subsubsection{Impulsantwort}
    \begin{itemize}
        \item Sei $a_m\ne 0$ und $y$ die Lösung (Output) der Differentialgleichung $L[y](t) = a_my^{(m)} + a_{m-1}y^{(m-1)}+...+a_1y'+a_0y=x$ mit $x \; T$-periodischer Input.
              Sei $P(ik\omega) = a_m(ik\omega)^m + ... + a_1(ik\omega) + a_0\ne 0$.
              Dann ist $x \mapsto y$ ein $T$-periodisches LTI-System mit Frequenzgang $d_k=\frac{1}{a_m(ik\omega)^m + ... + a_1(ik\omega) + a_0}$ und Impulsantwort $h_T(t)=\sum_{k\in\mathbb Z} d_ke^{ik\omega t}$
        \item Sei $L[h](t) = 0, P(ik\omega) \ne 0$, $h$ bel. oft stetig diffb. und \begin{align*}
                  h(0^+) = h(T^-)                 \\
                  h'(0^+) = h'(T^-)               \\
                  \vdots                          \\
                  h^{(m-2)}(0^+) = h^{(m-2)}(T^-) \\
                  h^{(m-1)}(0^+) = h^{(m-1)}(T^-) + \frac{T}{a_m}
              \end{align*}. Dann ist $h_T$ die Impulsantwort des LTI-Systems \\
              \textbf{Beispiel} \\
              \begin{enumerate}
                  \item System gegeben durch \begin{equation*}
                            \frac{1}{4}y'(t) + y(t) = x(t)
                        \end{equation*}
                  \item Umschreiben zu $L[h](t)$: \begin{equation*}
                            \frac{1}{4}h'(t) + h(t) = 0
                        \end{equation*}
                  \item Diese DGL Lösen $\Rightarrow h(t) = ce^{-4t}$
                  \item $h$ ist bel. oft stetig diffb. und $m=1$. Daraus folgt \begin{equation*}
                            h(0^+)=c \quad h(2\pi^-)=ce^{-8\pi} \Rightarrow c=ce^{-8\pi}+\frac{T}{a_1}
                        \end{equation*}
                  \item Auflösen nach $c$ ergibt die Impulsantwort \begin{equation*}
                            h(t) = \underbrace{\frac{8\pi}{1-e^{-8\pi}}}_c e^{-4t}
                        \end{equation*}
              \end{enumerate}
    \end{itemize}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Resonanz}
    $P(ik\omega) = 0$ für bestimmte $k\in\mathbb Z$
    \begin{itemize}
        \item Im Resonanzfall gibt es $T$-periodische Inputs $x$, die in nicht $T$-periodischen Outputs $y$ resultieren
        \item Keine eindeutige Lösung
    \end{itemize}
\end{sectionbox}


\subsection{Hilberträume}
\begin{emphbox}
    Ein (komplexer) Hilbertraum $\mathcal H$ ist ein $\mathbb C^n$-Vektorraum mit Skalarprodukt $\langle \cdot, \cdot \rangle : \mathcal H \times \mathcal H \rightarrow \mathbb C$, so dass $\mathcal H$ bzgl. der induzierten Norm $||v|| = \sqrt{\langle v,v\rangle}$ vollständig ist.
\end{emphbox}

\begin{emphbox}
    \subsubsection{Skalarprodukt im $\mathbb C^n$-Vektorraum $\mathcal V$}
    \begin{itemize}
        \item Linearität im 2. Argument \begin{equation*}
                  \langle v, \alpha u + \beta w\rangle = \alpha\langle v,u\rangle + \beta\langle v,w\rangle
              \end{equation*}
        \item Hermitizität \begin{equation*}
                  \langle v,w\rangle = \overline{\langle w,v\rangle}
              \end{equation*}
        \item Positive Definitheit \begin{equation*}
                  \langle v,v\rangle > 0 \; \forall v\in\mathcal{V}\setminus\{0\} \qquad \langle v,v\rangle = 0 \Leftrightarrow v=0
              \end{equation*}
    \end{itemize}
\end{emphbox}

\begin{emphbox}
    \subsubsection{Prä-Hilbertraum stückweise stetiger Funktionen}
    Sei $T>0$. Der Vektorraum
    \begin{equation*}
        \mathcal V := \{f:[0,T]\rightarrow \mathbb C | \text{ f stückweise stetig }\}
    \end{equation*}
    mit dem Skalarprodukt
    \begin{equation*}
        \langle f, g\rangle := \frac{1}{T}\int_0^T \overline{f(t)}g(t) \diff t
    \end{equation*}
    ist ein nicht-vollständiger (Prä-)Hilbertraum
\end{emphbox}

\subsection{Orthonormalsysteme}
\begin{sectionbox}
    Die Funktionen $e_k(t)=e^{ik\omega t}$ bilden im Raum der komplexen Funktionen ein Orthonormalsystem, da gleichzeitig
    \begin{equation*}
        \langle e_j,e_k\rangle = \delta_{j,k}
    \end{equation*}

    \subsubsection{Folgerungen}
    \begin{itemize}
        \item $\forall n \in \mathbb N_0, \forall \{\gamma_k\}_{k=-n}^{k=n} \subset\mathbb C$ gilt
              \begin{equation*}
                  \left|\left| \sum_{k=-n}^{n}\gamma_ke_k \right|\right|^2 = \sum_{k=-n}^{n}|\gamma_k|^2
              \end{equation*}
        \item $\forall f \in \mathcal{H}$ und jede endliche Teilmenge $\Omega \subset\mathbb Z$ gilt
              \begin{equation*}
                  \sum_{j\in\Omega}|\langle e_j, f\rangle|^2 \le ||f||^2
              \end{equation*}
        \item Für $f\in\mathcal{H}$ sei
              \begin{equation*}
                  S_f^n=\sum_{k=-n}^{n}c_ke_k \qquad c_k=\langle e_k, f\rangle.
              \end{equation*} Dann gilt
              \begin{equation*}
                  ||f||^2 = ||S_f^n||^2 + ||f-S_f^n||^2
              \end{equation*}
    \end{itemize}

    \subsubsection{Vollständigkeit}
    Ein Orthonormalsystem und Hilbertraum gilt als vollständig, wenn gilt
    \begin{equation*}
        \langle e_k, f \rangle = 0 \forall k \in \mathbb Z \Longrightarrow f = 0
    \end{equation*}
    
    \textbf{Äquivalente Aussagen}
    \begin{itemize}
        \item $\{e_n\}_{n\in\mathbb Z}$ ist ein vollständiges Orthonormalsystem.
        \item \begin{equation*}
                f = \lim_{n\rightarrow\infty}\sum_{k=-n}^{n}\langle e_k, f\rangle e_k \quad \forall f \in \mathcal H
            \end{equation*}
        \item Es gilt die Parseval-Identität 
            \begin{equation*}
                ||f||^2 = \sum_{k\in\mathbb Z}|\langle e_k, f\rangle|^2
            \end{equation*}
    \end{itemize}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Fourierreihenkonvergenz im quadratischen Mittel}
    Für vollständige Orthonormalsysteme gilt, dass jede stückweise stetige, $T$-periodische Funktion 
    \begin{equation*}
        \lim_{n\rightarrow\infty}\frac{1}{T}\int_0^T|f(t) - \sum_{k=-n}^{n}c_ke^{ik\omega t}|^2\diff t = 0
    \end{equation*}
    Nach Parseval-Bessel-Gleichung gilt dann
    \begin{equation*}
        \sum_{k\in\mathbb Z}|c_k|^2 = \frac{1}{T}\int_0^T|f(t)|^2\diff t
    \end{equation*}
    Fourierreihen, die diese Bedingung erfüllen, sind eindeutig. Falls $f,g$ das identische Fourierspektrum haben, so gilt $f(t)=g(t)$
\end{sectionbox}


\subsection{Fouriertransformation auf $\mathbb R$}
